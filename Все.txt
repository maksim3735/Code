    imageUrl: formGameImage.value,
    totalHours: Number(formTotalHours.value) || 0,

    startDate: formStartDate.value,
    rating: rawRating !== "" ? Number(rawRating) : null,
    sessionCount: Number(formSessionCount.value) || 0,

  const validation = validateGameForm(formData);

  if (!validation.isValid) {
    Object.keys(validation.errors).forEach((key) => {
      const errEl = document.getElementById(`err-${key}`);
      if (errEl) errEl.textContent = validation.errors[key];
    });
    return;
  }

  const calculatedTotalMinutes = Math.round(formData.totalHours * 60 + formData.totalMinutes);
  const isEditing = Boolean(formGameId.value);

  if (isEditing) {
    games = games.map((g) => {
      if (g.id !== formGameId.value) return g;
      return {
        ...g,
        name: formData.name.trim(),
        imageUrl: formData.imageUrl.trim(),
        totalTime: calculatedTotalMinutes,
        startDate: formData.startDate,
        rating: formData.rating,
        sessionCount: formData.sessionCount,
      };
    });

    const newGame: Game = {
      id: "game_" + Date.now().toString(),
      name: formData.name.trim(),
      imageUrl: formData.imageUrl.trim(),
      totalTime: calculatedTotalMinutes,

      sessionCount: formData.sessionCount,

import type { Game, ActiveSession } from "../types/game.types";

const GAMES_STORAGE_KEY = "game_tracker_games";
const ACTIVE_SESSION_STORAGE_KEY = "game_tracker_active_session";


export function loadGames(): Game[] {
  try {
    const rawData = localStorage.getItem(GAMES_STORAGE_KEY);
    if (!rawData) {
      return [];
    }
    return JSON.parse(rawData) as Game[];
  } catch (error) {
    console.error("Помилка при зчитуванні списку ігор зі сховища:", error);
    return [];
  }
}


export function saveGames(games: Game[]): void {
  try {
    const rawData = JSON.stringify(games);
    localStorage.setItem(GAMES_STORAGE_KEY, rawData);
  } catch (error) {
    console.error("Помилка при збереженні списку ігор у сховище:", error);
  }
}


export function loadActiveSession(): ActiveSession | null {
  try {
    const rawData = localStorage.getItem(ACTIVE_SESSION_STORAGE_KEY);
    if (!rawData) {
      return null;
    }
    return JSON.parse(rawData) as ActiveSession;
  } catch (error) {
    console.error("Помилка при зчитуванні активної сесії:", error);
    return null;
  }
}


export function saveActiveSession(session: ActiveSession): void {
  try {
    const rawData = JSON.stringify(session);

  } catch (error) {
    console.error("Помилка при збереженні активної сесії:", error);
  }
}

import type { ValidationResult, GameFormData } from "../types/game.types";


export function validateGameForm(data: GameFormData): ValidationResult {
  const errors: Record<string, string> = {};

  const trimName = data.name.trim();
  if (!trimName) {
    errors["name"] = "Назва гри є обов'язковою для заповнення.";
  } else if (trimName.length < 2) {
    errors["name"] = "Назва повинна містити мінімум 2 символи.";
  } else if (trimName.length > 60) {
    errors["name"] = "Назва не може перевищувати 60 символів.";
  }


  const trimmedUrl = data.imageUrl.trim();
  if (trimmedUrl.length > 0) {
    const isUrl = trimmedUrl.startsWith("http://") || trimmedUrl.startsWith("https://");
    if (!isUrl) {
      errors["imageUrl"] = "Введіть коректне посилання, що починається з http:// або https://";
    }
  }


  if (isNaN(data.totalHours) || data.totalHours < 0) {
    errors["totalHours"] = "Кількість годин не може бути від'ємною.";
  }

  if (isNaN(data.totalMinutes) || data.totalMinutes < 0 || data.totalMinutes > 59) {
    errors["totalMinutes"] = "Хвилини повинні бути в діапазоні від 0 до 59.";
  }


  if (data.rating !== null) {
    if (isNaN(data.rating) || data.rating < 1 || data.rating > 10) {
      errors["rating"] = "Оцінка повинна бути числом від 1 до 10.";
    }
  }


if (isNaN(data.sessionCount) || data.sessionCount < 0 || !Number.isInteger(data.sessionCount)) {
    errors["sessionCount"] = "Кількість сесій має бути цілим невід'ємним числом.";
  }


if (data.startDate) {
    const inputDate = new Date(data.startDate);
    const today = new Date();
    today.setHours(23, 59, 59, 999);

    const minDate = new Date("1970-01-01");

    if (isNaN(inputDate.getTime())) {
      errors["startDate"] = "Некоректний формат дати.";
    } else if (inputDate > today) {
      errors["startDate"] = "Дата початку гри не може бути пізнішою за сьогоднішню.";
    } else if (inputDate < minDate) {
      errors["startDate"] = "Дата початку гри не може бути ранішою за 01.01.1970.";
    }
  }

  return {
    isValid: Object.keys(errors).length === 0,
    errors,
  };
}

import type { Game } from "../types/game.types";


export function calculateNewRating(
  currentRating: number | null,
  sessionCount: number,
  newScore: number
): number {
  if (currentRating === null || sessionCount === 0) {
    return Math.round(newScore * 10) / 10;
  }

  const updatedRating = (currentRating * sessionCount + newScore) / (sessionCount + 1);
  return Math.round(updatedRating * 10) / 10;
}


export function formatMinutes(minutes: number): string {
  if (minutes <= 0) {
    return "0 хв";
  }

  const hours = Math.floor(minutes / 60);
  const remainingMinutes = minutes % 60;

  if (hours === 0) {
    return `${remainingMinutes} хв`;
  }

  if (remainingMinutes === 0) {
    return `${hours} год`;
  }

  return `${hours} год ${remainingMinutes} хв`;
}


export function isGameActual(lastSessionDate: string | null): boolean {
  if (!lastSessionDate) {
    return false;
  }

  const sessionDate = new Date(lastSessionDate);
  if (isNaN(sessionDate.getTime())) {
    return false;
  }

  const now = new Date();
  const diffTimeMs = now.getTime() - sessionDate.getTime();
  const diffDays = diffTimeMs / (1000 * 60 * 60 * 24);

  return diffDays >= 0 && diffDays <= 7;
}


export function getTodayDateString(): string {
  const today = new Date();
  const year = today.getFullYear();
  const month = String(today.getMonth() + 1).padStart(2, "0");
  const day = String(today.getDate()).padStart(2, "0");
  return `${year}-${month}-${day}`;
}

export function formatDateToDisplay(dateStr: string | null): string {
  if (!dateStr) {
    return "-";
  }

  const cleanDate = dateStr.includes("T") ? dateStr.split("T")[0] : dateStr;
  const parts = cleanDate.split("-");

  if (parts.length !== 3) {
    return dateStr;
  }

  const [year, month, day] = parts;
  return `${day}.${month}.${year}`;
}


export function getTopGamesData(games: Game[]): { labels: string[]; data: number[] } {
  if (games.length === 0) {
    return { labels: [], data: [] };
  }

  const sorted = [...games].sort((a, b) => b.totalTime - a.totalTime);

  if (sorted.length <= 5) {
    return {
      labels: sorted.map((g) => g.name),
      data: sorted.map((g) => g.totalTime),
    };
  }

  const top5 = sorted.slice(0, 5);
  const other = sorted.slice(5);
  const otherTotalTime = other.reduce((s, g) => s + g.totalTime, 0);

  return {
    labels: [...top5.map((g) => g.name), "Інші"],
    data: [...top5.map((g) => g.totalTime), otherTotalTime],
  };
}

export function formatTimerDisplay(totalSeconds: number): string {
  const hours = Math.floor(totalSeconds / 3600);
  const minutes = Math.floor((totalSeconds % 3600) / 60);
  const seconds = totalSeconds % 60;

  const h = String(hours).padStart(2, "0");
  const m = String(minutes).padStart(2, "0");
  const s = String(seconds).padStart(2, "0");

  return `${h}:${m}:${s}`;
}


export function getSessionSeconds(session: ActiveSession): number {  //чиста тривалість сесії без урахування пауз
  const now = Date.now(); 
  let pauseTime = session.allPauseTime;

  if (session.paused && session.pauseStartTime) {
    pauseTime += now - session.pauseStartTime;
  }

  const timeMs = now - session.startTime - pauseTime;
  return Math.max(0, Math.floor(timeMs / 1000));
}


export function getSessionMinutes(session: ActiveSession): number {
  const seconds = getSessionSeconds(session);
  return Math.max(1, Math.round(seconds / 60));
}


export function gameSelectList(
  selectEl: HTMLSelectElement,
  games: Game[]
): void {
  selectEl.innerHTML = "";

  const defaultOption = document.createElement("option");
  defaultOption.value = "";
  defaultOption.textContent = "-- Оберіть гру зі списку --";
  defaultOption.disabled = true; 
  defaultOption.selected = true; 
  selectEl.appendChild(defaultOption);

  games.forEach((g) => {
    const opt = document.createElement("option");
    opt.value = g.id;
    opt.textContent = g.name;
    selectEl.appendChild(opt);
  });
}


export function updateSessionUI(
  session: ActiveSession | null,
  games: Game[],
  elements: {
    panel: HTMLElement;
    noGamesBlock: HTMLElement;
    selectBlock: HTMLElement;
    activeBlock: HTMLElement;
    timerText: HTMLElement;
    statusText: HTMLElement;
    btnPause: HTMLButtonElement;
    gameTitle: HTMLElement;
  }
): void {

  if (!session) {
    elements.activeBlock.classList.add("hidden");

    if (games.length === 0) {
      elements.noGamesBlock.classList.remove("hidden");
      elements.selectBlock.classList.add("hidden");
    } else {
      elements.noGamesBlock.classList.add("hidden");
      elements.selectBlock.classList.remove("hidden");
    }
    return;
  }

  elements.noGamesBlock.classList.add("hidden");
  elements.selectBlock.classList.add("hidden");
  elements.activeBlock.classList.remove("hidden");

  const currentGame = games.find((g) => g.id === session.gameId);
  elements.gameTitle.textContent = currentGame ? currentGame.name : "Невідома гра";

  if (session.paused) {
    elements.statusText.textContent = "Сесія на паузі";
    elements.btnPause.textContent = "Продовжити";
    elements.btnPause.className = "btn btn-success";
  } else {
    elements.statusText.textContent = "Сесія триває...";
    elements.btnPause.textContent = "Призупинити";
    elements.btnPause.className = "btn btn-warning";
  }
}

import type { Game, LibraryFilter, LibrarySort } from "../types/game.types";
import { formatMinutes, formatDateToDisplay, isGameActual } from "../utils/calculations";

import defaultPosterImg from "../assets/void.jpg"; 

const DEFAULT_POSTER = defaultPosterImg;

export function filterAndSortGames(
  games: Game[],
  search: string,
  filterType: LibraryFilter, 
  sortBy: LibrarySort
): Game[] {


  let result = [...games];


  const slovo = search.trim().toLowerCase();
  if (slovo.length > 0) {
    result = result.filter((g) => g.name.toLowerCase().includes(slovo));
  }

  
  if (filterType === "favorites") {
    result = result.filter((g) => g.favorite);
  } else if (filterType === "actual") {
    result = result.filter((g) => isGameActual(g.lastSessionDate));
  }


  if (sortBy === "name") {
    result.sort((a, b) => a.name.localeCompare(b.name, "uk"));
  } else if (sortBy === "lastSessionDate") {
    result.sort((a, b) => {
      const timeA = a.lastSessionDate ? new Date(a.lastSessionDate).getTime() : 0;
      const timeB = b.lastSessionDate ? new Date(b.lastSessionDate).getTime() : 0;
      return timeB - timeA;
    });
  } else if (sortBy === "createdAt") {
    result.sort((a, b) => b.id.localeCompare(a.id));
  }

  return result;
}


export function renderGameCards(
  games: Game[],
  container: HTMLElement,
  emptyMessageEl: HTMLElement,
  callbacks: {
    onToggleFavorite: (id: string) => void;
    onEdit: (game: Game) => void;
    onDelete: (id: string) => void;
  }
): void {
  container.innerHTML = "";

  if (games.length === 0) {
    emptyMessageEl.classList.remove("hidden");
    return;
  }

  emptyMessageEl.classList.add("hidden");

  games.forEach((game) => {
    const card = document.createElement("div");
    card.className = "game-card";

    const posterUrl = game.imageUrl && game.imageUrl.trim().length > 0 
      ? game.imageUrl : DEFAULT_POSTER;

    const ratingText = game.rating !== null ? `★ ${game.rating}` : "Без оцінки";
    const lastPlayedText = formatDateToDisplay(game.lastSessionDate);

    card.innerHTML = `
      <div class="card-poster-wrapper">
        <img 
          src="${posterUrl}" 
          alt="${game.name}" 
          class="card-poster"
          onerror="this.src='${DEFAULT_POSTER}'"
        />
        <button 
          class="btn-favorite ${game.favorite ? "active" : ""}" 
          title="${game.favorite ? "Видалити з улюблених" : "Додати в улюблені"}"
        >
          ${game.favorite ? "❤️" : "🤍"}
        </button>
      </div>
      <div class="card-body">
        <h3 class="card-title" title="${game.name}">${game.name}</h3>
        <div class="card-meta">
          <span>Час: <strong>${formatMinutes(game.totalTime)}</strong></span>
          <span>Оцінка: <strong>${ratingText}</strong></span>
        </div>
        <div class="card-meta">
          <span>Остання гра: <strong>${lastPlayedText}</strong></span>
          <span>Сесій: <strong>${game.sessionCount}</strong></span>
        </div>
        <div class="card-actions">
          <button class="btn btn-secondary btn-edit">Редагувати</button>
          <button class="btn btn-danger btn-delete">Видалити</button>
        </div>
      </div>
    `;

    const btnFav = card.querySelector(".btn-favorite") as HTMLButtonElement;
    btnFav.addEventListener("click", () => callbacks.onToggleFavorite(game.id));

    const btnEdit = card.querySelector(".btn-edit") as HTMLButtonElement;
    btnEdit.addEventListener("click", () => callbacks.onEdit(game));

    const btnDelete = card.querySelector(".btn-delete") as HTMLButtonElement;
    btnDelete.addEventListener("click", () => callbacks.onDelete(game.id));

    container.appendChild(card);
  });
}

import type { Game, TableSort, SortDirection } from "../types/game.types";
import { formatMinutes, formatDateToDisplay } from "../utils/calculations";


export function sortTableGames(
  games: Game[],
  sortField: TableSort,
  direction: SortDirection
): Game[] {
  const result = [...games];

  result.sort((a, b) => {

    let x = 0;  // Значення для порівняння

    switch (sortField) {

      case "name":
        x = a.name.localeCompare(b.name, "uk");
        break;

      case "totalTime":
        x = a.totalTime - b.totalTime;
        break;

      case "startDate": {
        const timeA = a.startDate ? new Date(a.startDate).getTime() : 0;
        const timeB = b.startDate ? new Date(b.startDate).getTime() : 0;
        x = timeA - timeB;
        break;
      }

      case "rating": {
        const ratingA = a.rating ?? -1;
        const ratingB = b.rating ?? -1;
        x = ratingA - ratingB;
        break;
      }

      case "sessionCount":
        x = a.sessionCount - b.sessionCount;
        break;

      case "lastSessionDate": {
        const timeA = a.lastSessionDate ? new Date(a.lastSessionDate).getTime() : 0;
        const timeB = b.lastSessionDate ? new Date(b.lastSessionDate).getTime() : 0;
        x = timeA - timeB;
        break;
      }

      case "lastSessionTime":
        x = a.lastSessionTime - b.lastSessionTime;
        break;
    }

    return direction === "up" ? x : -x;
  });

  return result;
}


export function renderTable(
  games: Game[],
  currentPage: number,
  tbodyEl: HTMLElement,
  paginationInfoEl: HTMLElement,
  btnPrev: HTMLButtonElement,
  btnNext: HTMLButtonElement
): void {

  tbodyEl.innerHTML = "";

  const totalPages = Math.ceil(games.length / 10) || 1;
  const safePage = Math.min(Math.max(currentPage, 1), totalPages);

  const startIndex = (safePage - 1) * 10;
  const pageGames = games.slice(startIndex, startIndex + 10);

  if (pageGames.length === 0) {
    const emptyRow = document.createElement("tr");
    emptyRow.innerHTML = `<td colspan="7" class="empty-table-cell">Дані відсутні</td>`;
    tbodyEl.appendChild(emptyRow);
  } else {
    pageGames.forEach((game) => {
      const row = document.createElement("tr");

      const ratingDisplay = game.rating !== null ? `★ ${game.rating}` : "-";
      const startDateDisplay = formatDateToDisplay(game.startDate);
      const lastDateDisplay = formatDateToDisplay(game.lastSessionDate);
      const lastSessionDisplay = game.lastSessionTime > 0 ? formatMinutes(game.lastSessionTime) : "-";

      row.innerHTML = `
        <td><strong>${game.name}</strong></td>
        <td>${formatMinutes(game.totalTime)}</td>
        <td>${startDateDisplay}</td>
        <td>${ratingDisplay}</td>
        <td>${game.sessionCount}</td>
        <td>${lastDateDisplay}</td>
        <td>${lastSessionDisplay}</td>
      `;

      tbodyEl.appendChild(row);
    });
  }

  paginationInfoEl.textContent = `Сторінка ${safePage} з ${totalPages}`;
  btnPrev.disabled = safePage <= 1;
  btnNext.disabled = safePage >= totalPages;
}


export function updateSortIcons(
  headerCells: NodeListOf<HTMLTableCellElement>,
  currentSort: TableSort,
  currentDirection: SortDirection
): void {
  headerCells.forEach((th) => {
    const field = th.dataset.sort as TableSort | undefined;
    const iconSpan = th.querySelector(".sort-icon");

    if (!iconSpan) return;

    if (field === currentSort) {
      iconSpan.textContent = currentDirection === "up" ? " ▲" : " ▼";
    } else {
      iconSpan.textContent = "";
    }
  });
}

import { Chart, registerables } from "chart.js";
import type { Game } from "../types/game.types";
import { getTopGamesData } from "../utils/calculations";

Chart.register(...registerables);

let timeChart: Chart | null = null;
let sessionsChart: Chart | null = null;


export function updateStatsSummary(
  games: Game[],
  totalTimeEl: HTMLElement,
  gamesCountEl: HTMLElement,
  sessionsCountEl: HTMLElement
): void {
  const totalMinutes = games.reduce((sum, g) => sum + g.totalTime, 0);
  const totalHours = Math.round((totalMinutes / 60) * 10) / 10;
  const totalSessions = games.reduce((sum, g) => sum + g.sessionCount, 0);

  totalTimeEl.textContent = `${totalHours} год`;
  gamesCountEl.textContent = games.length.toString();
  sessionsCountEl.textContent = totalSessions.toString();
}


export function renderTimeChart(canvas: HTMLCanvasElement, games: Game[]): void {

  const { labels, data } = getTopGamesData(games);

  if (timeChart) {
    timeChart.destroy();
  }

  const hoursData = data.map((min) => Math.round((min / 60) * 10) / 10);
  timeChart = new Chart(canvas, {
    type: "pie",
    data: {
      labels: labels.length > 0 ? labels : ["Немає даних"],
      datasets: [
        {
          data: hoursData.length > 0 ? hoursData : [1],
          backgroundColor: [
            "#0609bd",
            "#ba29f4",
            "#ff0080",
            "#10b981",
            "#f59e0b",
            "#64748b",
          ],
          borderColor: "#11111c",
          borderWidth: 2,
        },
      ],
    },
    options: {
      responsive: true,
      maintainAspectRatio: true,
      plugins: {
        legend: {
          position: "bottom",
          labels: {
            color: "#94a3b8",
            font: { size: 12 },
          },
        },
      },
    },
  });
}


export function renderSessionsChart(canvas: HTMLCanvasElement, games: Game[]): void {

  if (sessionsChart) {
    sessionsChart.destroy();
  }

  const top10 = [...games]
    .sort((a, b) => b.sessionCount - a.sessionCount)
    .slice(0, 10);

  const labels = top10.map((g) => g.name);
  const data = top10.map((g) => g.sessionCount);

  sessionsChart = new Chart(canvas, {
    type: "bar",
    data: {
      labels: labels.length > 0 ? labels : ["Немає даних"],
      datasets: [
        {
          label: "Кількість сесій",
          data: data.length > 0 ? data : [0],
          backgroundColor: "#0070f3",
          borderRadius: 6,
        },
      ],
    },
    options: {
      responsive: true,
      maintainAspectRatio: true,
      scales: {
        x: {
          ticks: { color: "#94a3b8" },
          grid: { display: false },
        },
        y: {
          beginAtZero: true,
          ticks: {
            color: "#94a3b8",
            stepSize: 1,
          },
          grid: { color: "#2b2f45" },
        },
      },
      plugins: {
        legend: { display: false },
      },
    },
  });
}
